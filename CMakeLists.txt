cmake_minimum_required(VERSION 3.22)
project(omarchy_replay VERSION 0.1.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
find_package(Qt6 REQUIRED COMPONENTS Core Gui Widgets Test Concurrent DBus Network)
find_package(tomlplusplus CONFIG REQUIRED)
find_package(OpenMP REQUIRED COMPONENTS CXX)
find_package(PkgConfig REQUIRED)
pkg_check_modules(NATIVE REQUIRED IMPORTED_TARGET tesseract lept sqlite3 libwebp wayland-client)
pkg_get_variable(WAYLAND_PROTOCOLS wayland-protocols pkgdatadir)
find_program(WAYLAND_SCANNER wayland-scanner REQUIRED)

set(PROTOCOL_SOURCES)
foreach(PROTOCOL ext-image-copy-capture-v1 ext-image-capture-source-v1 ext-foreign-toplevel-list-v1)
    string(REGEX REPLACE "-v1$" "" PROTOCOL_DIR "${PROTOCOL}")
    set(XML "${WAYLAND_PROTOCOLS}/staging/${PROTOCOL_DIR}/${PROTOCOL}.xml")
    set(HEADER "${CMAKE_CURRENT_BINARY_DIR}/${PROTOCOL}-client-protocol.h")
    set(CODE "${CMAKE_CURRENT_BINARY_DIR}/${PROTOCOL}-protocol.c")
    add_custom_command(OUTPUT "${HEADER}" "${CODE}"
        COMMAND ${WAYLAND_SCANNER} client-header "${XML}" "${HEADER}"
        COMMAND ${WAYLAND_SCANNER} private-code "${XML}" "${CODE}"
        DEPENDS "${XML}" VERBATIM)
    list(APPEND PROTOCOL_SOURCES "${CODE}" "${HEADER}")
endforeach()

add_library(replay_core STATIC src/recorder.cpp src/fixture.cpp src/viewer.cpp src/work_budget.cpp src/index_service.cpp
    src/replay_config.cpp src/recording_control.cpp src/recording_environment.cpp src/agent_prompt.cpp src/storage_forecast.cpp src/selection_ocr.cpp
    src/meeting_index.cpp src/meeting_view.cpp)
target_include_directories(replay_core PUBLIC src)
target_link_libraries(replay_core PUBLIC Qt6::Widgets Qt6::Concurrent Qt6::Network Qt6::DBus PkgConfig::NATIVE OpenMP::OpenMP_CXX tomlplusplus::tomlplusplus)
target_compile_options(replay_core PRIVATE -Wall -Wextra -Wpedantic)
add_executable(replay src/main.cpp src/recording_service.cpp src/capture.cpp ${PROTOCOL_SOURCES})
target_compile_definitions(replay PRIVATE REPLAY_VERSION="${PROJECT_VERSION}")
target_include_directories(replay PRIVATE "${CMAKE_CURRENT_BINARY_DIR}")
target_link_libraries(replay PRIVATE replay_core)
target_compile_options(replay PRIVATE -Wall -Wextra -Wpedantic)
